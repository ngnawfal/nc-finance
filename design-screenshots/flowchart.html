2. Alur Login & Role Guard

flowchart TD
  A["Buka aplikasi"] --> B{"Token valid
di localStorage?"}
  B -- "Tidak" --> C["Redirect /login"]
  C --> D["Isi email + password"]
  D --> E["POST /api/auth/login"]
  E --> F{"Backend verifikasi"}
  F -- "Gagal" --> G["Tampilkan error"] --> D
  F -- "Sukses" --> H["Simpan nc_token + nc_user"]
  H --> I["Redirect /dashboard"]
  B -- "Ya" --> J["GET /api/auth/me"]
  J --> K{"Token aktif?"}
  K -- "Tidak" --> C
  K -- "Ya" --> L["Render layout sesuai role"]
  L --> M{"Role?"}
  M -- "Staf" --> N["Sidebar tanpa Validasi & Pengaturan"]
  M -- "Manajer" --> O["Sidebar lengkap"]

3. Alur Pendaftaran Peserta

flowchart TD
  P1["Staf buka /pendaftaran"] --> P2["GET /api/pengaturan/kursus"]
  P2 --> P3["Isi data peserta + pilih kursus"]
  P3 --> P4{"Tombol mana?"}
  P4 -- "Simpan Pendaftaran" --> P5["statusPendaftaran=terdaftar"]
  P4 -- "Simpan Draft" --> P6["statusPendaftaran=draft"]
  P5 --> P7["POST /api/pendaftaran"]
  P6 --> P7
  P7 --> P8["Backend insert
pendaftaran_peserta"]
  P8 --> P9["Toast sukses + reset form"]
  P9 --> P10["Muncul di /siswa & /berkas"]

4. Alur Transaksi (Input → Validasi → Laporan)

flowchart TD
  T1["Staf buka /transaksi/input"] --> T2["Pilih peserta, nominal, metode, bukti"]
  T2 --> T3["POST /api/transaksi"]
  T3 --> T4["Backend simpan
statusValidasi=pending"]
  T4 --> T5["Masuk antrian validasi"]
  T5 --> T6["Notifikasi pending di Topbar"]
  T6 --> T7["Manajer buka /transaksi/validasi"]
  T7 --> T8{"Approve / Reject?"}
  T8 -- "Approve" --> T9["PUT /api/transaksi/{id}/validate
status=approved"]
  T8 -- "Reject" --> T10["PUT /api/transaksi/{id}/validate
status=rejected + alasan"]
  T9 --> T11["Update tabel transaksi"]
  T10 --> T11
  T11 --> T12["Masuk perhitungan
/dashboard & /laporan"]
  T12 --> T13{"Hanya approved
yang dihitung"}

flowchart TD
